cmake_minimum_required(VERSION 3.14)
project(local_inference_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "$ORIGIN")
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_library(li_bridge SHARED bridge.cpp)
target_link_libraries(li_bridge PRIVATE llama ggml)
target_include_directories(li_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(li_bridge PRIVATE -Wall -Wextra -Werror)
add_executable(li-reference reference.cpp)
target_link_libraries(li-reference PRIVATE li_bridge)
target_compile_options(li-reference PRIVATE -Wall -Wextra -Werror)
